Persona 5 Royal — Phantom Thieves 
lua

-- ~/.config/nvim/lua/plugins/persona5.lua
return {
  {
    "folke/tokyonight.nvim",
    lazy = false,
    priority = 1000,
    opts = {
      style = "night",
      on_colors = function(c)
        -- P5R: Stark black, blood red, pure white, gold
        c.bg           = "#0D0D0D"  -- Velvet Room black
        c.bg_dark      = "#000000"  -- Pure void
        c.bg_highlight = "#1A0000"  -- Dark crimson shadow
        c.fg           = "#FFFFFF"  -- Phantom white
        c.red          = "#FF0000"  -- Morgana red / Joker mask
        c.red1         = "#CC0000"
        c.orange       = "#FF4500"  -- Flame accent
        c.yellow       = "#FFD700"  -- Gold trim
        c.green        = "#00FF41"  -- Matrix-ish terminal green
        c.cyan         = "#FF1493"  -- Hot pink UI glitch
        c.blue         = "#FF0033"  -- Red-shifted blue
        c.blue1        = "#C0392B"
        c.blue5        = "#FF6B6B"
        c.magenta      = "#FF0055"  -- Phantom red-magenta
        c.comment      = "#555555"  -- Grayed-out shadows
        c.border       = "#FF0000"  -- Red borders, very P5
      end,
      on_highlights = function(hl, c)
        hl.CursorLine      = { bg = "#1A0000" }
        hl.Visual          = { bg = "#3D0000" }
        hl.Search          = { bg = "#FF0000", fg = "#000000", bold = true }
        hl.IncSearch       = { bg = "#FFD700", fg = "#000000", bold = true }
        hl.LineNr          = { fg = "#FF0000" }
        hl.CursorLineNr    = { fg = "#FFD700", bold = true }
        hl.FloatBorder     = { fg = "#FF0000", bg = "#0D0D0D" }
        hl.NormalFloat     = { bg = "#0A0000" }
        hl.StatusLine      = { bg = "#FF0000", fg = "#000000", bold = true }
        hl.TabLineSel      = { bg = "#FF0000", fg = "#FFFFFF", bold = true }
        hl.WinSeparator    = { fg = "#FF0000" }
        -- AI agent / LSP diagnostics
        hl.DiagnosticError = { fg = "#FF0000", bold = true }
        hl.DiagnosticWarn  = { fg = "#FFD700" }
        hl.DiagnosticInfo  = { fg = "#FF6B6B" }
        hl.DiagnosticHint  = { fg = "#FF1493" }
      end,
    },
    config = function(_, opts)
      require("tokyonight").setup(opts)
      vim.cmd("colorscheme tokyonight")
    end,
  },
}

Switching Between Themes

Add this to your lazy.nvim setup to toggle between them easily:
lua

-- In your keymaps.lua
vim.keymap.set("n", "<leader>cp5", function()
  -- swap your plugin file and re-source, or just:
  require("tokyonight").load({ style = "night" }) -- after setting P5 opts
end, { desc = "Persona 5 Royal theme" })

Or use a theme picker like themery.nvim to hot-swap between all three.

Each scheme is tuned for AI agent workflows — DiagnosticError/Warn/Info/Hint are all styled distinctly so LLM tool output, LSP errors, and inline hints stay visually separated per each game's palette.
